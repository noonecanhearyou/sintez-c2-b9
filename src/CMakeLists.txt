cmake_minimum_required(VERSION 3.21)
project(LibreGenesis VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Widgets Concurrent Test)
set(BUILD_TESTING OFF CACHE BOOL "Upstream tests" FORCE)
add_subdirectory(homology EXCLUDE_FROM_ALL)
add_library(synteny synteny/src/synteny.cpp)
target_include_directories(synteny PUBLIC synteny/include)
target_link_libraries(synteny PUBLIC homology Qt6::Core)
add_library(orf_core orf/src/parser.cpp orf/src/scanner.cpp orf/src/alignment.cpp orf/src/evidence.cpp orf/src/profile.cpp orf/src/pipeline.cpp)
target_include_directories(orf_core PUBLIC orf/include orf/third_party)
target_link_libraries(orf_core PUBLIC Qt6::Core)
add_library(geneorigin_core app/src/core.cpp app/src/pipeline.cpp app/src/denovo.cpp)
target_include_directories(geneorigin_core PUBLIC app/include)
target_link_libraries(geneorigin_core PUBLIC homology synteny orf_core Qt6::Core)
target_compile_definitions(geneorigin_core PUBLIC GENEORIGIN_WORKSPACE="${CMAKE_SOURCE_DIR}")
add_library(geneorigin_ui app/src/window.cpp app/src/tree.cpp app/include/window.hpp app/include/tree.hpp)
target_link_libraries(geneorigin_ui PUBLIC geneorigin_core Qt6::Widgets Qt6::Concurrent)
add_executable(LibreGenesis MACOSX_BUNDLE app/src/main.cpp app/resources/resources.qrc)
target_link_libraries(LibreGenesis PRIVATE geneorigin_ui)
set_target_properties(LibreGenesis PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "LibreGenesis" MACOSX_BUNDLE_GUI_IDENTIFIER "org.libregenesis.studio" MACOSX_BUNDLE_SHORT_VERSION_STRING "1.0.0")
option(GENEORIGIN_TESTS "Build application tests" ON)
if(GENEORIGIN_TESTS)
 enable_testing()
 add_executable(geneorigin-tests app/tests/tests.cpp)
 target_link_libraries(geneorigin-tests PRIVATE geneorigin_core Qt6::Test)
 add_test(NAME application-core COMMAND geneorigin-tests)
 add_executable(geneorigin-ui-tests app/tests/gui_tests.cpp)
 target_link_libraries(geneorigin-ui-tests PRIVATE geneorigin_ui Qt6::Test)
 add_test(NAME application-ui COMMAND geneorigin-ui-tests)
 set_tests_properties(application-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS LibreGenesis BUNDLE DESTINATION . RUNTIME DESTINATION bin)
