cmake_minimum_required(VERSION 3.21)
project(Homology VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core)
set(CMAKE_AUTOMOC ON)
add_library(homology src/homology.cpp src/prepare.cpp src/batch.cpp src/job.cpp include/homology/job.hpp)
target_include_directories(homology PUBLIC include)
target_link_libraries(homology PUBLIC Qt6::Core)
target_compile_options(homology PRIVATE -Wall -Wextra -Wpedantic)
add_executable(homology-cli src/main.cpp)
target_link_libraries(homology-cli PRIVATE homology)
include(CTest)
if(BUILD_TESTING)
 add_executable(homology-tests tests/tests.cpp)
 target_link_libraries(homology-tests PRIVATE homology)
 add_test(NAME homology-tests COMMAND homology-tests)
endif()
if(BUILD_TESTING)
 add_executable(homology-validate tests/validate_real.cpp)
 target_link_libraries(homology-validate PRIVATE homology)
endif()
if(BUILD_TESTING)
 add_executable(homology-audit tests/audit_all.cpp)
 target_link_libraries(homology-audit PRIVATE homology)
endif()
